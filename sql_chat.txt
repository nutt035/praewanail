-- 1. ตารางเก็บข้อมูลผู้ใช้ (แยก LINE และ Facebook)
CREATE TABLE chat_users (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  platform TEXT NOT NULL, -- 'line' หรือ 'facebook'
  platform_user_id TEXT NOT NULL UNIQUE,
  display_name TEXT,
  picture_url TEXT,
  last_active TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 2. ตารางเก็บสถานะแชท (Bot / Human Handoff)
CREATE TABLE chat_sessions (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  chat_user_id UUID REFERENCES chat_users(id) ON DELETE CASCADE,
  status TEXT DEFAULT 'bot', -- 'bot' หรือ 'human'
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  UNIQUE(chat_user_id)
);

-- 3. ตารางเก็บประวัติข้อความ (Message Logs)
CREATE TABLE chat_messages (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  chat_user_id UUID REFERENCES chat_users(id) ON DELETE CASCADE,
  direction TEXT NOT NULL, -- 'inbound' (ลูกค้าพิมพ์มา) หรือ 'outbound' (เรา/บอทตอบกลับ)
  message_type TEXT DEFAULT 'text', -- 'text', 'image', 'sticker'
  content TEXT,
  image_url TEXT,
  delivery_status TEXT DEFAULT 'success', -- 'success', 'error', 'pending'
  error_message TEXT,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- ==========================================
-- RLS (Row Level Security) สำหรับ Admin (ถ้าจำเป็น)
-- ==========================================
-- อนุญาตให้ทุกคนใน role authenticated / service_role อ่านและเขียนได้
ALTER TABLE chat_users ENABLE ROW LEVEL SECURITY;
ALTER TABLE chat_sessions ENABLE ROW LEVEL SECURITY;
ALTER TABLE chat_messages ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Allow ALL on chat_users" ON chat_users FOR ALL USING (true);
CREATE POLICY "Allow ALL on chat_sessions" ON chat_sessions FOR ALL USING (true);
CREATE POLICY "Allow ALL on chat_messages" ON chat_messages FOR ALL USING (true);
